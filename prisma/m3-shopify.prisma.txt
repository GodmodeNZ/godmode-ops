// Staging schema fragment for M3. Merge into schema.prisma before migration.

enum SalesOrderStatus { RECEIVED READY_FOR_PRODUCTION BLOCKED IN_PRODUCTION COMPLETED CANCELLED }
enum SalesOrderLineStatus { UNRESOLVED RESOLVED BLOCKED }
enum IntegrationEventStatus { RECEIVED PROCESSED FAILED IGNORED }

model SalesOrder {
  id String @id @default(cuid())
  source String
  externalId String
  orderNumber String
  shopDomain String?
  financialStatus String?
  fulfillmentStatus String?
  currency String?
  total Decimal? @db.Decimal(12,2)
  customerName String?
  customerEmail String?
  status SalesOrderStatus @default(RECEIVED)
  externalCreatedAt DateTime?
  raw Json?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  lines SalesOrderLine[]
  @@unique([source, externalId])
  @@index([source, orderNumber])
  @@index([status, createdAt])
}

model SalesOrderLine {
  id String @id @default(cuid())
  salesOrderId String
  salesOrder SalesOrder @relation(fields:[salesOrderId], references:[id], onDelete:Cascade)
  externalLineId String
  title String
  sku String?
  shopifyProductId String?
  shopifyVariantId String?
  quantity Int
  properties Json?
  status SalesOrderLineStatus @default(UNRESOLVED)
  resolutionMessage String?
  mappingId String?
  resolvedProductId String?
  buildIds String[] @default([])
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  @@unique([salesOrderId, externalLineId])
  @@index([status])
}

model ShopifyProductMapping {
  id String @id @default(cuid())
  shopDomain String?
  shopifyProductId String?
  shopifyVariantId String?
  sku String?
  productId String
  active Boolean @default(true)
  priority Int @default(100)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  configurationRules ShopifyConfigurationRule[]
  @@index([shopDomain, shopifyVariantId, active])
  @@index([shopDomain, shopifyProductId, active])
  @@index([shopDomain, sku, active])
}

model ShopifyConfigurationRule {
  id String @id @default(cuid())
  mappingId String
  mapping ShopifyProductMapping @relation(fields:[mappingId], references:[id], onDelete:Cascade)
  propertyName String
  propertyValue String
  role String
  replacementSkuId String
  quantity Int?
  createdAt DateTime @default(now())
  @@index([mappingId, propertyName, propertyValue])
}

model IntegrationEvent {
  id String @id @default(cuid())
  provider String
  externalEventId String
  topic String
  shopDomain String?
  status IntegrationEventStatus @default(RECEIVED)
  triggeredAt DateTime?
  processedAt DateTime?
  error String?
  payload Json?
  createdAt DateTime @default(now())
  @@unique([provider, externalEventId])
  @@index([provider, topic, createdAt])
}
